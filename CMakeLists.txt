cmake_minimum_required(VERSION 3.16)
project(GutC VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
add_library(gutc STATIC src/gutc.c src/json.c src/json_dom.c src/url_query.c src/auth_contracts.c src/oauth_contracts.c src/string_case.c
  src/gutc_path.c src/trusted_api.c src/collections.c src/safe_path.c src/identity.c
  src/gut_string_predicates.c src/gut_truncate.c src/gut_truncate_words.c)
add_library(GutC::gutc ALIAS gutc)
set_target_properties(gutc PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_include_directories(gutc PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(MSVC)
  target_compile_options(gutc PRIVATE /W4 /WX)
else()
  target_compile_options(gutc PRIVATE -Wall -Wextra -Werror -Wpedantic)
  if(UNIX AND NOT APPLE)
    target_link_libraries(gutc PUBLIC m)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(gutc_test tests/test_gutc.c)
  target_link_libraries(gutc_test PRIVATE GutC::gutc)
  target_compile_definitions(gutc_test PRIVATE
    GUTC_FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/testdata/upstream-golden/identity.tsv")
  add_test(NAME gutc_foundation COMMAND gutc_test)
  add_executable(gutc_json_test tests/test_json.c)
  target_link_libraries(gutc_json_test PRIVATE GutC::gutc)
  target_compile_definitions(gutc_json_test PRIVATE
    GUTC_JSON_FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/testdata/upstream-golden/json.tsv")
  add_test(NAME gutc_json COMMAND gutc_json_test)
  add_executable(gutc_collections_test tests/test_collections.c)
  target_link_libraries(gutc_collections_test PRIVATE GutC::gutc)
  add_test(NAME gutc_collections COMMAND gutc_collections_test)
  add_executable(gutc_safe_path_test tests/test_safe_path.c)
  target_link_libraries(gutc_safe_path_test PRIVATE GutC::gutc)
  add_test(NAME gutc_safe_path COMMAND gutc_safe_path_test)
  add_executable(gutc_identity_test tests/test_identity.c)
  target_link_libraries(gutc_identity_test PRIVATE GutC::gutc)
  add_test(NAME gutc_identity COMMAND gutc_identity_test)
  add_executable(gutc_cross_language_test tests/test_cross_language.c)
  target_link_libraries(gutc_cross_language_test PRIVATE GutC::gutc)
  target_compile_definitions(gutc_cross_language_test PRIVATE
    GUTC_GOLDEN_FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/testdata/upstream-golden/contracts.tsv")
  add_test(NAME gutc_cross_language COMMAND gutc_cross_language_test)
  add_executable(gutc_json_dom_test tests/test_json_dom.c)
  target_link_libraries(gutc_json_dom_test PRIVATE GutC::gutc)
  add_test(NAME gutc_json_dom COMMAND gutc_json_dom_test)
  add_executable(gutc_url_query_test tests/test_url_query.c)
  target_link_libraries(gutc_url_query_test PRIVATE GutC::gutc)
  add_test(NAME gutc_url_query COMMAND gutc_url_query_test)
  add_executable(gutc_auth_contracts_test tests/test_auth_contracts.c)
  target_link_libraries(gutc_auth_contracts_test PRIVATE GutC::gutc)
  target_compile_definitions(gutc_auth_contracts_test PRIVATE
    GUTC_AUTH_FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/testdata/upstream-golden/auth.tsv")
  add_test(NAME gutc_auth_contracts COMMAND gutc_auth_contracts_test)
  add_executable(gutc_oauth_contracts_test tests/test_oauth_contracts.c)
  target_link_libraries(gutc_oauth_contracts_test PRIVATE GutC::gutc)
  target_compile_definitions(gutc_oauth_contracts_test PRIVATE
    GUTC_OAUTH_FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/testdata/upstream-golden/oauth.tsv")
  add_test(NAME gutc_oauth_contracts COMMAND gutc_oauth_contracts_test)
  add_executable(gutc_strings_test tests/test_string_case.c)
  target_link_libraries(gutc_strings_test PRIVATE GutC::gutc)
  target_compile_definitions(gutc_strings_test PRIVATE
    GUTC_STRINGS_FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/testdata/upstream-golden/strings.tsv")
  add_test(NAME gutc_strings COMMAND gutc_strings_test)
  foreach(pair IN ITEMS "path;test_path.c" "trusted_api;trusted_api_test.c" "string_predicates;test_string_predicates.c" "truncate;test_truncate.c" "truncate_words;test_truncate_words.c")
    list(GET pair 0 name)
    list(GET pair 1 file)
    add_executable(gutc_${name}_test tests/${file})
    target_link_libraries(gutc_${name}_test PRIVATE GutC::gutc)
    add_test(NAME gutc_${name} COMMAND gutc_${name}_test)
  endforeach()
  foreach(check_target IN ITEMS gutc_test gutc_json_test gutc_json_dom_test gutc_url_query_test gutc_auth_contracts_test gutc_oauth_contracts_test gutc_strings_test gutc_collections_test gutc_safe_path_test gutc_identity_test gutc_cross_language_test gutc_path_test

    if(MSVC)
      target_compile_options(${check_target} PRIVATE /UNDEBUG)

      target_compile_options(${check_target} PRIVATE -UNDEBUG)

install(TARGETS gutc EXPORT GutCTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
configure_package_config_file(cmake/GutCConfig.cmake.in
  "${CMAKE_CURRENT_BINARY_DIR}/GutCConfig.cmake"
  INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/GutC")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/GutCConfigVersion.cmake"
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/GutCConfig.cmake"
  "${CMAKE_CURRENT_BINARY_DIR}/GutCConfigVersion.cmake"
  DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/GutC")
install(EXPORT GutCTargets NAMESPACE GutC:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/GutC")
