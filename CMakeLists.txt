cmake_minimum_required(VERSION 3.16)
project(gutc_trusted_api C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(gutc_trusted_api STATIC src/trusted_api.c)
target_include_directories(gutc_trusted_api PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(gutc_trusted_api PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
add_executable(gutc_trusted_api_test tests/trusted_api_test.c)
target_link_libraries(gutc_trusted_api_test PRIVATE gutc_trusted_api)
add_test(NAME trusted_api COMMAND gutc_trusted_api_test)
