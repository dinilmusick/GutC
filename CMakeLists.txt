cmake_minimum_required(VERSION 3.16)
project(GutC LANGUAGES C)
add_library(gutc_truncate STATIC src/gut_truncate.c src/gut_truncate_words.c src/gut_string_predicates.c)
target_include_directories(gutc_truncate PUBLIC include)
set_target_properties(gutc_truncate PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
include(CTest)
if(BUILD_TESTING)
 add_executable(gutc_truncate_test tests/test_truncate.c)
 target_link_libraries(gutc_truncate_test PRIVATE gutc_truncate)
 add_test(NAME gutc_truncate_test COMMAND gutc_truncate_test)
 add_executable(gutc_truncate_words_test tests/test_truncate_words.c)
 target_link_libraries(gutc_truncate_words_test PRIVATE gutc_truncate)
 add_test(NAME gutc_truncate_words_test COMMAND gutc_truncate_words_test)
 add_executable(gutc_string_predicates_test tests/test_string_predicates.c)
 target_link_libraries(gutc_string_predicates_test PRIVATE gutc_truncate)
 add_test(NAME gutc_string_predicates_test COMMAND gutc_string_predicates_test)
 # C assert() disappears when NDEBUG is defined (e.g. Release builds).
 # Keep regression checks executable in every CI build configuration.
 foreach(test_target IN ITEMS gutc_truncate_test gutc_truncate_words_test gutc_string_predicates_test)
  if(MSVC)
   target_compile_options(${test_target} PRIVATE /UNDEBUG)
  else()
   target_compile_options(${test_target} PRIVATE -UNDEBUG)
  endif()
 endforeach()
endif()
