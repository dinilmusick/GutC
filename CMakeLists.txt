cmake_minimum_required(VERSION 3.12)
project(GutC LANGUAGES C)
add_library(gutc_url STATIC src/url.c)
target_include_directories(gutc_url PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
set_target_properties(gutc_url PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
include(CTest)
if(BUILD_TESTING)
  add_executable(gutc_url_test tests/url_test.c)
  target_link_libraries(gutc_url_test PRIVATE gutc_url)
  add_test(NAME gutc_url_test COMMAND gutc_url_test)
endif()
